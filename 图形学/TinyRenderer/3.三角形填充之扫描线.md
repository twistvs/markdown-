通过改变y来取x范围的思想，非常类似于==二重积分==，也正因如此，三角形需要分成上下两个部分

#### 扫描线绘制三角形

##### 基本思想

按照纵坐标y，将画面分为一根根水平的扫描线，一帧渲染一根扫描线上的内容，需要做的操作便是判断当前扫描线的哪一部分在三角形内

##### 第一步：将三个顶点按照y坐标从大到小排序

```c++
void triangle(int ax, int ay, int bx, int by, int cx, int cy, TGAImage & framebuffer, TGAColor color)
{
	//冒泡排序，把点按y坐标从小到大排序
    if (ay > by)
    {
        std::swap(ax, bx);
        std::swap(ay, by);
    }
    if (ay > cy)
    {
        std::swap(ax, cx);
        std::swap(ay, cy);
    }
    if (by > cy)
    {
        std::swap(bx, cx);
        std::swap(by, cy);
    }
	//最高的点和最低的点连成的边为长边，另外两个为短边
    line(ax, ay, bx, by, framebuffer, green);
    line(bx, by, cx, cy, framebuffer, green);
    line(cx, cy, ax, ay, framebuffer, red);
}
```

当扫描线在y方向上下移动时，他在长边上的比例是线性的，而这个比例可以相应的映射出当前扫描线与两条边的交点。

由于有两条短边，所以需要沿着b点用水平线把三角形一分为二，分别进行上下两部分的渲染

##### 第二步：下半三角形

下半三角形在ay与by之间，所以采样也应该由y坐标主导，通过y坐标求出与左右两条边的交点的x坐标

```c++
void triangle(int ax, int ay, int bx, int by, int cx, int cy, TGAImage & framebuffer, TGAColor color)
{
	//冒泡排序，把点按y坐标从小到大排序
    if (ay > by)
    {
        std::swap(ax, bx);
        std::swap(ay, by);
    }
    if (ay > cy)
    {
        std::swap(ax, cx);
        std::swap(ay, cy);
    }
    if (by > cy)
    {
        std::swap(bx, cx);
        std::swap(by, cy);
    }

    int total_height = cy - ay;
	int low_height = by - ay;
	int high_height = cy - by;
    for (int y = ay; y <= by; y++)
    {
        int x_long, x_short;
		x_long = ax + (cx - ax) * (y - ay) / total_height;
        x_short = ax + (bx - ax) * (y - ay) / low_height;
        framebuffer.set(x_short, y, color);
		framebuffer.set(x_long, y, color);
    }
}
```

##### 第三步：下半三角形填充

填充只需要在每个y上都把x_long和x_short中间的点都遍历一遍即可

```c++
{

    }

    {
        std::swap(bx, cx);
        std::swap(by, cy);

	int low_height = by - ay;

		framebuffer.set(x_long, y, color);

        {
			std::swap(x_long, x_short);
        }
        for (int x = x_short; x <= x_long; x++)
        {
			framebuffer.set(x, y, color);
        }
    }
}
```

##### 第四步：上半三角形填充

```c++
void triangle(int ax, int ay, int bx, int by, int cx, int cy, TGAImage & framebuffer, TGAColor color)
{
	//冒泡排序，把点按y坐标从小到大排序
    if (ay > by)
    {
        std::swap(ax, bx);
        std::swap(ay, by);
    }
    if (ay > cy)
    {
        std::swap(ax, cx);
        std::swap(ay, cy);
    }
    if (by > cy)
    {
        std::swap(bx, cx);
        std::swap(by, cy);
    }

    int total_height = cy - ay;
	int low_height = by - ay;
	int high_height = cy - by;
    int x_long, x_short;
    for (int y = ay; y <= by; y++)
    {
        
		x_long = ax + (cx - ax) * (y - ay) / total_height;
        x_short = ax + (bx - ax) * (y - ay) / low_height;
        framebuffer.set(x_short, y, color);
		framebuffer.set(x_long, y, color);
        if (x_long < x_short)
        {
			std::swap(x_long, x_short);
        }
        for (int x = x_short; x <= x_long; x++)
        {
			framebuffer.set(x, y, color);
        }
    }
    for (int y = by; y <= cy; y++)
    {
        x_long = cx + (ax - cx) * (cy - y) / total_height;
        x_short = bx + (cx - bx) * (y - by) / high_height;
        if (x_long < x_short)
        {
            std::swap(x_long, x_short);
        }
        for (int x = x_short; x <= x_long; x++)
        {
            framebuffer.set(x, y, color);
        }
    }
}
```
