#### 1.重心坐标插值灰度

灰度的有效范围是0-255，而alpha * az + beta * bz + gamma * cz的范围也是0-255，所以可以直接当作灰度使用

```c++
//灰度图，利用z坐标作为深度值

#include <cmath>
#include "../tgaimage.h"

double signed_triangle_area(int ax, int ay, int bx, int by, int cx, int cy) {
    return .5 * ((by - ay) * (bx + ax) + (cy - by) * (cx + bx) + (ay - cy) * (ax + cx));
}

void triangle(int ax, int ay, int az, int bx, int by, int bz, int cx, int cy, int cz, TGAImage& framebuffer) {
    int bbminx = std::min(std::min(ax, bx), cx); // bounding box for the triangle
    int bbminy = std::min(std::min(ay, by), cy); // defined by its top left and bottom right corners
    int bbmaxx = std::max(std::max(ax, bx), cx);
    int bbmaxy = std::max(std::max(ay, by), cy);
    double total_area = signed_triangle_area(ax, ay, bx, by, cx, cy);
    if (total_area < 1) return; // backface culling + discarding triangles that cover less than a pixel

	//CPU多核并行计算for循环
#pragma omp parallel for
    for (int x = bbminx; x <= bbmaxx; x++) {
        for (int y = bbminy; y <= bbmaxy; y++) {
            double alpha = signed_triangle_area(x, y, bx, by, cx, cy) / total_area;
            double beta = signed_triangle_area(x, y, cx, cy, ax, ay) / total_area;
            double gamma = signed_triangle_area(x, y, ax, ay, bx, by) / total_area;
            if (alpha < 0 || beta < 0 || gamma < 0) continue; // negative barycentric coordinate => the pixel is outside the triangle
            //重心坐标插值深度
            unsigned char z = static_cast<unsigned char>(alpha * az + beta * bz + gamma * cz);
			//set的第三个参数是TGAColor类型，传入{z}会调用TGAColor的构造函数，创建一个bgra[4] = {z,0,0,0}的color对象
            framebuffer.set(x, y, { z });
        }
    }
}

int main(int argc, char** argv) {
    constexpr int width = 64;
    constexpr int height = 64;
	//bpp传入GRAYSCALE=1，在set函数中memcpy也只会复制data的一字节
    TGAImage framebuffer(width, height, TGAImage::GRAYSCALE);

    int ax = 17, ay = 4, az = 13;
    int bx = 55, by = 39, bz = 128;
    int cx = 23, cy = 59, cz = 255;

    triangle(ax, ay, az, bx, by, bz, cx, cy, cz, framebuffer);

    framebuffer.write_tga_file("framebuffer.tga");
    return 0;
}
```

```c++
void triangle(int ax, int ay, int bx, int by, int cx, int cy, TGAImage& framebuffer)
{
    int minx, maxx, miny, maxy;
    minx = std::min(ax, std::min(bx, cx));
    maxx = std::max(ax, std::max(bx, cx));
    miny = std::min(ay, std::min(by, cy));
    maxy = std::max(ay, std::max(by, cy));
    //三角形的总面积
    double total_area = signed_triangle_area(ax, ay, bx, by, cx, cy);

    TGAColor color;
    for (int x = minx; x <= maxx; x++) {
        for (int y = miny; y <= maxy; y++) {
            //计算重心坐标
            double alpha = signed_triangle_area(x, y, bx, by, cx, cy) / total_area;
            double beta = signed_triangle_area(x, y, cx, cy, ax, ay) / total_area;
            double gamma = signed_triangle_area(x, y, ax, ay, bx, by) / total_area;
            //重心坐标全为正才说明点在三角形内部
            if (alpha < 0 || beta < 0 || gamma < 0) continue; // negative barycentric coordinate => the pixel is outside the triangle
			//通过将重心坐标限制在0.1以下，可以只渲染靠近顶点的区域，也就是一个线框
            if (alpha < 0.1 || beta < 0.1 || gamma < 0.1)
            {
                color.bgra[0] = alpha * blue.bgra[0] + beta * green.bgra[0] + gamma * red.bgra[0];
                color.bgra[1] = alpha * blue.bgra[1] + beta * green.bgra[1] + gamma * red.bgra[1];
                color.bgra[2] = alpha * blue.bgra[2] + beta * green.bgra[2] + gamma * red.bgra[2];
                framebuffer.set(x, y, color);
            }
        }
    }
}

```

重心坐标代表了点P距离三角形顶点的距离，通过限制P与顶点之间的距离，就可以实现绘制边框的效果
