```c++
#include <iostream>
#include <vector>
#include <algorithm>
#include <string>
#include <map>

using namespace std;

//棋盘类，存储棋子，Add方法传入棋子，棋子基类，两个棋子的子类，两种不同的棋子
//基类属性：id,z-order,show()打印，素材名称Name
class ChessBase {
private:
    int id;
    int zOrder;
    string name;

public:
    void Setid(int NewId) {
        id = NewId;
    }
    void SetZ(int NewZ) {
        zOrder = NewZ;
    }
    void SetName(string NewName) {
        name = NewName;
    }

    int GetId() {
        return id;
    }
    int GetZ() {
        return zOrder;
    }
    string GetName() {
        return name;
    }
};

class ChessA : public ChessBase{
public:
    ChessA() {
        Setid(1);
        SetZ(2);
        SetName("ChessA");
    }
};

class ChessB : public ChessBase{
public:
    ChessB() {
        Setid(2);
        SetZ(1);
        SetName("ChessB");
    }
};

class ChessPlate {
private:
    //为什么不存引用或者传右值
    multimap<int, ChessBase*> chessMap;

public:
    void AddChess(ChessBase* chess) {
        chessMap.emplace(chess->GetZ(), chess);
    }

    void show() {
        for (auto i : chessMap) {
            cout << i.second->GetName() << endl;
        }
    }
};

int main() {
    ChessA chess1;
    ChessA chess2;
    ChessB chess3;
    ChessB chess4;

    ChessPlate plate;

    plate.AddChess(&chess1);
    plate.AddChess(&chess2);
    plate.AddChess(&chess3);
    plate.AddChess(&chess4);

    plate.show();


    return 0;
}
```
