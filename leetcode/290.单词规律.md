***很坑的题***

​	本质上就是个char和string的双向哈希，但是他不给你string数组，让你自己从string里面拆出来

​	他题目里头不跟你说char和string的数量必须得一样，实际上你必须得保证两者数量一致

```c++
class Solution {
public:
    bool wordPattern(string pattern, string s) {
        vector<string> v;

        int left = 0;
        int len = s.size();

        //用双指针来把所有的string拆出来
        for (int right = 0; right < len; ++right) {
            if (s[right] == ' ') {
                v.push_back(s.substr(left, right - left));
                left = right + 1;
            }
        }

        // 加入最后一个单词
        v.push_back(s.substr(left));

        if (v.size() != pattern.size()) {
            return false;
        }

        unordered_map<char, string> m1;
        unordered_map<string, char> m2;

        for (int i = 0; i < pattern.size(); ++i) {
            if (m1.count(pattern[i])) {
                if (m1[pattern[i]] != v[i]) {
                    return false;
                }
            }
            else {
                m1[pattern[i]] = v[i];
            }

            if (m2.count(v[i])) {
                if (m2[v[i]] != pattern[i]) {
                    return false;
                }
            }
            else {
                m2[v[i]] = pattern[i];
            }
        }

        return true;
    }
};
```
