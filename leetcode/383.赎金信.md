- 字母计数数组做计数，ransomNote的计数必须小于magazine才行

```c++
class Solution {
public:
    bool canConstruct(string ransomNote, string magazine) {
        //字母计数数组
        vector<int> count(26, 0);

        for(char c : magazine){
            count[c - 'a']++;
        }

        //如果有一个字母的count被减到0以下，说明肯定不满足条件
        for(char c : ransomNote){
            count[c - 'a']--;
            if(count[c - 'a'] < 0){
                return false;
            } 
        }

        return true;
    }
};
```
