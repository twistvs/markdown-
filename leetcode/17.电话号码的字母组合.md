#### 回溯

- 外部顺序为数字的顺序，如先从2 "abc"选，再从3 "def"选
- 内部顺序为字母选择的顺序，对于2的外部循环来说，他是从0到abc.size() - 1

```c++
class Solution {
public:
    string temp;
    vector<string> ans;
    //start是指先选择pick中的哪一个数字（由于答案可以按任意顺序返回），所以顺序是可以随意设置的，我们让他按照从0开始的顺序进行选择
    void backtrack(vector<string>& pick, int start){
        //如果temp和pick一样长，则此时已经选完，插入ans
        if(temp.size() == pick.size()){
            ans.push_back(temp);
            return;
        }

        //第一选从abc中选，先选a，再选b，最后选c
        for(int i = 0; i < pick[start].size(); i++){
            temp += pick[start][i];
            //第一步走完，第一步是从pick[start + 1]，即"def"中选
            backtrack(pick, start + 1);
            temp.pop_back();
        }
    }

    vector<string> letterCombinations(string digits) {
        vector<string> vec = {"", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};
        vector<string> pick;
        //pick里面装了digits对应的字符串["abc", "def"]
        for(char digit : digits){
            pick.push_back(vec[digit - '0']);
        }
        backtrack(pick, 0);
        return ans;
    }
};
```

#### 回溯的标准格式

```c++
//有些题目并不一定有步数的限制，但是一定会有别的限制，而这个限制就是需要传的参数。这一步的选择列表并不需要直接传参，完全可以在函数中进行解析
void backtrack(这是第几步？, 这一步有哪些选择？) {  
    if (满足结束条件) {   //一般情况下，这个结束条件都是由这是第几步？来限制的
        保存路径;
        return;
    }
	
    //遍历这一步的所有选择
    for (选择 : 选择列表) {
        做选择;

        //在这一步执行某个选择的情况下走下一步，需要根据实际情况来确定下一步的选择列表，有些题目中，上一步的选择会影响下一步的选择列表
        backtrack(...);

        撤销选择;
    }
}
```

#### 优化过的结构

```c++
class Solution {
public:

    vector<string> ans;

    string temp;

    //我并没有直接把当前步的选择列表传参，而是把所有的选择列表传参，用当前步数step来获取当前步骤的选择列表
    void backtrack(int step, vector<string>& input){
        //结束条件
        if(step >= input.size()){
            ans.push_back(temp);
            return;
        }

        int len = input[step].size();

        //遍历当前步骤的选择列表
        for(int i = 0; i < len; i++){
            temp.push_back(input[step][i]);

            //走下一步
            backtrack(step + 1, input);

            temp.pop_back();
        }
    }
    
    vector<string> letterCombinations(string digits) {
        vector<string> v = {"", "", "abc", "def","ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};

        vector<string> input;

        for(char digit : digits){
            input.push_back(v[digit - '0']);
        }

        backtrack(0, input);

        return ans;
    }
};
```
