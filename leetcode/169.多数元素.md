向下取整函数floor

最简单的方法就是用哈希表，数量大于一半的元素就是答案

```c++
class Solution {
public:
    int majorityElement(vector<int>& nums) {
        unordered_map<int, int> m;

        int len = nums.size();

        int half = floor(len / 2);

        for(int num : nums){
            m[num]++;
            if(m[num] > half){
                return num;
            }
        }

        return 0;
    }
};
```
