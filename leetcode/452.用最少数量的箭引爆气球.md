#### 排序+贪心

​	排序之后，一次取交集，最后剩几个区间就需要几箭

​	*贪心体现在哪里*？我们每遍历到一个气球时，都会去试图找到能一次射爆最多气球的点，这就是贪心

```c++
class Solution {
public:
    int findMinArrowShots(vector<vector<int>>& points) {
        int n = points.size();

        sort(points.begin(), points.end());

        vector<vector<int>> v;

        for(int i = 0; i < n; i++){
            if(i == 0){
                v.push_back(points[i]);
            }
            else{
                int start1 = v.back()[0];
                int end1 = v.back()[1];
                int start2 = points[i][0];
                int end2 = points[i][1];
                if(start2 <= end1){
                    v.back()[0] = max(start1, start2);
                    v.back()[1] = min(end1, end2);
                }
                else{
                    v.push_back(points[i]);
                }
            }
        }

        return v.size();
    }
};
```
