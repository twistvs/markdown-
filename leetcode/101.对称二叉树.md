#### 少见的双递归

​	这个题并不是从root开始递归的，因为==root本身的状态和他是否是对称二叉树没有关系==

​	只要root->left和root->right是对称的，那么root就是对称的

```c++
class Solution {
public:
    bool mirror(TreeNode* left, TreeNode* right){
        //如果两个树都是空的，那么他们两个是对称的
        if(!left && !right){
            return true;
        }
        //如果只有一个树是空的，那么他们两个一定不是对称的
        else if(!left){
            return false;
        }
        else if(!right){
            return false;
        }

        //如果两个都不是空的，那么需要做判断
        int a = mirror(left->left, right->right);
        int b = mirror(left->right, right->left);

        if(left->val == right->val && a && b){
            return true;
        }
        else{
            return false;
        }
    }

    bool isSymmetric(TreeNode* root) {
        if(!root->left && !root->right){
            return true;
        }

        return mirror(root->left, root->right);
    }
};
```
