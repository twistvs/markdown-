​	没得啥技巧，就是要把newInterval插进去

​	由于intervals本来就是无重叠并且递增的，所以我们要先找到开始插newInterval的起点，然后再做逐个元素合并的操作，一直做到插newInterval的终点，起点之前和终点之后均不需要做区间合并的操作

```c++
class Solution {
public:
    vector<vector<int>> insert(vector<vector<int>>& intervals, vector<int>& newInterval) {
        int left = newInterval[0];
        int right = newInterval[1];
        bool placed = false;
        vector<vector<int>> ans;
        for (const auto& interval: intervals) {
            if (interval[0] > right) {
                // 在插入区间的右侧且无交集
                if (!placed) {
                    ans.push_back({left, right});
                    placed = true;                    
                }
                ans.push_back(interval);
            }
            else if (interval[1] < left) {
                // 在插入区间的左侧且无交集
                ans.push_back(interval);
            }
            else {
                // 与插入区间有交集，计算它们的并集
                left = min(left, interval[0]);
                right = max(right, interval[1]);
            }
        }
        //如果遍历到最后都没插，说明newInterval在最右边
        if (!placed) {
            ans.push_back({left, right});
        }
        return ans;
    }
};
```
