#### 依旧弗洛伊德判圈算法

```c++
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        ListNode* slow = head;
        ListNode* fast = head;

        if(!head || !head->next){
            return nullptr;
        }

        while(fast && fast->next){
            slow = slow->next;
            fast = fast->next->next;

            //走到一起之后直接进行回到开头走第二轮的操作，也可以写在while的外面
            if(slow == fast){
                slow = head;

                while(slow != fast){
                    slow = slow->next;
                    fast = fast->next;
                }

                return slow;
            }
        }
        return nullptr;
    }
};
```
