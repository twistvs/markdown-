**双指针**

和比target大，就让右指针往左挪；和比target小，就让左指针往右挪


```c++
class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        int n = numbers.size();

        int left = 0;
        int right = n - 1;

        while(left < right){
            int sum = numbers[left] + numbers[right];

            if(sum == target){
                break;
            }
            else if(sum > target){
                right--;
            }
            else{
                left++;
            }
        }

        return {left + 1, right + 1};
    }
};
```
