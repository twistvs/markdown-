#### 回溯

- 回溯的不同路径之间不是if,else的关系，而是都要走一遍的关系

```c++
class Solution {
public:
    void backtrack(vector<string>& ans, int n, string& s, int left, int right){
        if(left + right == n*2){
            ans.push_back(s);
            return;
        }
		
        //先左后右，虽然有if判定，但是每个路径都是需要走的
        if(left < n){
            s.push_back('(');
            backtrack(ans, n, s, left + 1, right);
            s.pop_back();
        }
        if(right < left){
            s.push_back(')');
            backtrack(ans, n, s, left, right + 1);
            s.pop_back();
        }

    }

    vector<string> generateParenthesis(int n) {
        vector<string> ans;
        string s = "";
        backtrack(ans, n, s, 0, 0);
        return ans;
    }
};
```

#### 优化过的格式

​	这个看起来要更长一些，但是实际逻辑要稍微清晰一点

```c++
class Solution {
public:
    //step是步数
    void backtrack(int step, int n) {
        if (step >= n * 2) {
            ans.push_back(temp);
            return;
        }
		
        //如果此时左括号已经大于一半，选择列表里面就只有右括号
        if (leftCount >= n) {
            temp += ')';
            rightCount++;
            backtrack(step + 1, n);
            rightCount--;
            temp.pop_back();
        } 
        else {
            //在左括号少于一半时，如果左括号比右括号多，那么选择列表里面两个括号都有，但是左括号要在前面
            if (leftCount > rightCount) {
                temp += '(';
                leftCount++;
                backtrack(step + 1, n);
                leftCount--;
                temp.pop_back();

                temp += ')';
                rightCount++;
                backtrack(step + 1, n);
                rightCount--;
                temp.pop_back();
            } 
            //如果左括号和右括号一样多，那么选择列表只有左括号
            else if (leftCount == rightCount) {
                temp += '(';
                leftCount++;
                backtrack(step + 1, n);
                leftCount--;
                temp.pop_back();
            }
        }
    }

    vector<string> generateParenthesis(int n) {
        backtrack(0, n);

        return ans;
    }

private:
    vector<string> ans;

    string temp;

    int leftCount = 0;
    int rightCount = 0;
};
```
