#### 前驱结点法

1. 先找到root的左子树的最右孩子
2. 把root的右子树放到这个孩子的右子树
3. 把root的左子树放到root的右子树
4. 清空root的左子树
5. root = root->right
6. 循环直到root等于NULL为止

```c++
class Solution {
public:
    void flatten(TreeNode* root) {
        TreeNode* curr = root;
        TreeNode* predececor = root;
        while (curr) {
            if (curr->left) {
                predececor = curr->left;
                while (predececor->right) {
                    predececor = predececor->right;
                }
                if (curr->right) {
                    predececor->right = curr->right;
                }
                curr->right = curr->left;
                curr->left = NULL;
            }
            curr = curr->right;
        }
    }
};
```

#### 相对简单一点的反先序遍历+头插法

```c++
class Solution {
private:
    TreeNode* head = nullptr;

    void dfs(TreeNode* root) {
        if (!root) {
            return;
        }

        // 先处理右子树，再处理左子树
        dfs(root->right);
        dfs(root->left);

        // 将当前节点插入链表头部
        root->right = head;
        root->left = nullptr;

        // 更新链表头节点
        head = root;
    }

public:
    void flatten(TreeNode* root) {
        head = nullptr;
        dfs(root);
    }
};
```
