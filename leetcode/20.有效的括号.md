#### 栈的最基本形态

​	把判断两个字符是否匹配写成一个辅助函数，会更方便一些

```c++
class Solution {
public:
    bool isMatch(char a, char b){
        if(a == '(' && b == ')') return true;
        if(a == '[' && b == ']') return true;
        if(a == '{' && b == '}') return true;

        return false;
    }

    bool isValid(string s) {
        stack<char> stk;

        for(char c : s){
            if(stk.empty()){
                stk.push(c);
            }
            else{
                if(isMatch(stk.top(), c)){
                    stk.pop();
                }
                else{
                    stk.push(c);
                }
            }
        }

        if(stk.empty()){
            return true;
        }
        else{
            return false;
        }
    }
};
```
