#### DFS

​	如果两个树相同，那么他们遍历的过程中所有节点的值都相等

​	root如果是相同树，那么他们的左子树和右子树都得是相同树，所以是后序遍历

```c++
class Solution {
public:
    bool dfs(TreeNode* p, TreeNode* q){
        if(!p && !q){
            return true;
        }
        else if(p && !q){
            return false;
        }
        else if(q && !p){
            return false;
        }

        bool a = dfs(p->left, q->left);
        bool b = dfs(p->right, q->right);

        //左子树和右子树都是相同树，并且root自己也相等
        if(a && b && p->val == q->val){
            return true;
        }
        else{
            return false;
        }
    }

    bool isSameTree(TreeNode* p, TreeNode* q) {
        return dfs(p, q);
    }
};
```
