#### <u>相当复杂的一个新建数据结构问题</u>

------

​	**如何记录关键字的常用顺序？**

容易想到list，最新访问的关键字放到队尾，如果capacity已满，就把访问的最少的队头弹出

​	**如果不用list，该怎么实现？**

list本质上就是个双向链表，所以我们要自己创建一个双向链表

------

1. 创建节点数据结构Listnode，成员应该包括key,val,prev,next
2. 创建unordered_map<key, Listnode*>  实现key值到实际节点的映射

```c++
struct Listnode{
    int key;
    int val;
    Listnode* prev;
    Listnode* next;

    Listnode(int key, int val){
        this->key = key;
        this->val = val;
        prev = NULL;
        next = NULL;
    }
};

class LRUCache {
public:
    //容量，key与节点的映射，双向链表的头节点和尾节点
    int capacity;
    unordered_map<int, Listnode*> mp;
    Listnode* head = new Listnode(0, 0);
    Listnode* tail = new Listnode(0, 0);

    //初始化，获得capacity的值，并且将head和tail连接起来
    LRUCache(int capacity) {
        this->capacity = capacity;
        head->next = tail;
        tail->prev = head;
    }
    
    int get(int key) {
        //如果没找到，返回-1
        if(mp.count(key) == 0){
            return -1;
        }
        //如果找到了，返回节点的值，并且将节点放到队尾
        else{
            moveToTail(mp[key]);
            return mp[key]->val;
        }
    }
    
    void put(int key, int value) {
        //如果map里面已经有这个key了，更新val值并且把key放到队尾
        if(mp.count(key)){
            mp[key]->val = value;
            moveToTail(mp[key]);
        }
        //如果map里面没有这个key，把key插入队尾
        else{
            Listnode* node = new Listnode(key, value);
            mp[key] = node;
            addToTail(node);
            
            //如果插入进来之后容量不够用了，要弹出队头
            if(mp.size() > capacity){
                Listnode* erased = popHead();
                mp.erase(erased->key);
                delete erased;
            }
        }
    }
	
    //挪到队尾的操作可以分解成删除后再添加到队尾
    void moveToTail(Listnode* node){
        deleteNode(node);
        addToTail(node);
    }

    void deleteNode(Listnode* node){
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    void addToTail(Listnode* node){
        tail->prev->next = node;
        node->prev = tail->prev;
        node->next = tail;
        tail->prev = node;
    }

    //由于我们要用key来访问map以删除map的对应项，所以需要把弹出的队头返回去
    Listnode* popHead(){
        Listnode* temp = head->next;
        deleteNode(temp);
        return temp;
    }
};
```

#### 二刷的理解

​	get的时间复杂度为O(1)，代表要用哈希表要做查找

​	要同时支持尾插和头出，还要支持把任意位置的元素搬到末尾（deque搞不了这个），要用双向链表list

**手动实现简易的双向链表**

​	双向链表的数据结构很简单，就是比单向链表多一个pre指针，主要是双向链表的构建需要两个dummy节点，并且需要在初始化时把他们两个连在一起

```c++
struct Node {
    int key;
    int val;
    Node* pre;
    Node* next;

    Node(int key, int val) : key(key), val(val) {}
};

class LRUCache {
public:
    // 记录现在链表有多少个元素
    int num = 0;

    int capacity;
    unordered_map<int, Node*> m;

    Node* preDummy = new Node(-1, 0);
    Node* lastDummy = new Node(-1, 0);

    LRUCache(int capacity) {
        this->capacity = capacity;
        preDummy->next = lastDummy;
        lastDummy->pre = preDummy;
    }

    int get(int key) {
        if (m.count(key)) {
            moveToTail(m[key]);
            return m[key]->val;
        } else {
            return -1;
        }
    }

    void put(int key, int value) {
        if (m.count(key)) {
            m[key]->val = value;
            moveToTail(m[key]);
        }
        // 如果m里不存在当前key，那么就需要分情况，如果capacity够，就直接插，否则要挤掉队首
        else {
            Node* temp = new Node(key, value);
            tailInsert(temp);
            m[key] = temp;
            num++;

            if (num > capacity)
            {
                Node* del = preDummy->next;
                m.erase(del->key);
                delNode(del);
                delete del;
                num--;
            }
        }
    }

    void tailInsert(Node* temp) {
        lastDummy->pre->next = temp;
        temp->next = lastDummy;
        temp->pre = lastDummy->pre;
        lastDummy->pre = temp;
    }

    void delNode(Node* del){
        Node* preNode = del->pre;
        Node* nextNode = del->next;
        preNode->next = nextNode;
        nextNode->pre = preNode;
    }

    void moveToTail(Node* move){
        delNode(move);
        tailInsert(move);
    }
};
```
