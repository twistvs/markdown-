#### 层次遍历

​	这道题可以作为层次遍历的标准格式

```c++
class Solution {
public:
    Node* connect(Node* root) {
        //处理输入为空的特殊情况
        if(root == nullptr){
            return root;
        }

        //队列，先把第一层加进来
        queue<Node*> q;
        q.push(root);

        //循环的总结束条件是队列为空
        while(!q.empty()){
            //依次遍历每一层，每次遍历之前把这一层的size记录下来
            int size = q.size();

            //根据这一层的size直接做遍历
            for(int i = 0; i < size; i++){
                //在让队头出队之前，先用temp记录下来
                Node* temp = q.front();
                q.pop();

                //用temp做cao'zu
                if(i == size - 1){
                    temp->next = nullptr;
                }
                else{
                    temp->next = q.front();
                }

                if(temp->left){
                    q.push(temp->left);
                }
                if(temp->right){
                    q.push(temp->right);
                }
            }
        }

        return root;
    }
};
```
