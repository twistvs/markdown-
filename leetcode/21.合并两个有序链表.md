#### 基础题

​	只有在list1和list2都遍历完的时候才停止继续插入

```c++
class Solution {
public:
    ListNode* mergeTwoLists(ListNode* list1, ListNode* list2) {
        ListNode* dummy = new ListNode(0);
        ListNode* curr = dummy;


        while(list1 || list2){
            if(list1 && list2){
                if(list1->val <= list2->val){
                    curr->next = new ListNode(list1->val);
                    curr = curr->next;
                    list1 = list1->next;
                }
                else{
                    curr->next = new ListNode(list2->val);
                    curr = curr->next;
                    list2 = list2->next;
                }
            }
            else if(list1){
                curr->next = new ListNode(list1->val);
                curr = curr->next;
                list1 = list1->next;
            }
            else{
                curr->next = new ListNode(list2->val);
                curr = curr->next;
                list2 = list2->next;
            }
        }

        return dummy->next;
    }
};
```
