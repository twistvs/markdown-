#### 回溯

- 这道题的回溯条件只有“索引不重复”，没有起始点等等别的条件
- 为了记录索引是否已经访问过，需要一个visited数组

```c++
class Solution {
public:
    void backtrack(vector<int>& nums, vector<int>& temp,
                   vector<vector<int>>& ans, int n, vector<bool>& visited) {
        if (temp.size() == n) {
            ans.push_back(temp);
            return;
        }

        for (int i = 0; i < n; i++) {
            if (visited[i] == false) {
                temp.push_back(nums[i]);
                visited[i] = true;
                backtrack(nums, temp, ans, n, visited);
                visited[i] = false;
                temp.pop_back();
            }
        }
    }

    vector<vector<int>> permute(vector<int>& nums) {
        int n = nums.size();
        vector<int> temp;
        vector<vector<int>> ans;
        vector<bool> visited(n, false);
        backtrack(nums, temp, ans, n, visited);
        return ans;
    }
};
```

#### 优化过的结构

```c++
class Solution {
public:
    vector<vector<int>> ans;

    vector<int> temp;

    //用一个表来记录当前索引前面是否已经走过了
    unordered_map<int, bool> m;

    //step是当前走到第几步。同样没有在传参时传入当前步骤的选择列表，而是传入总的选择列表
    void backtrack(int step, vector<int> & nums){
        //最多只能走nums.size()步
        if(step >= nums.size()){
            ans.push_back(temp);
            return;
        }

        //对于每一步来讲，他的选择列表=[0, n)去掉前面已经走过的索引
        //for循环给出了[0,n)
        for(int i = 0; i < nums.size(); i++){
            //剪枝剪掉了前面已经走过的索引，至此当前步骤的选择列表已经形成
            if(m[i]){
                continue;
            }

            //在走这一步的时候不要忘了把当前索引已经走过的标记更新一下
            m[i] = true;
            temp.push_back(nums[i]);
            //走下一步
            backtrack(step + 1, nums);
            temp.pop_back();
            m[i] = false;
        }
    }

    vector<vector<int>> permute(vector<int>& nums) {
        int n = nums.size();

        for(int i = 0; i < n; i++){
            m[i] = false;
        }

        backtrack(0, nums);

        return ans;
    }
};
```
