```c++
class Solution {
public:
    int lengthOfLastWord(string s) {
        int length = 0;

        int n = s.size();

        //从右边遍历，寻找最后一个单词的结尾；注意s可能全是空格，这样就会一路遍历到0
        int start = n - 1;
        while(s[start] == ' ' && start >= 0){
            start--;
        }

        //如果s不是全是空格，start最终会停在>=0的位置，从这里为起点反向遍历，获得最后一个单词的长度
        if(start >= 0){
            for(int i = start; i >= 0; i--){
                if(s[i] != ' '){
                    length++;
                }
                else{
                    break;
                }
            }
        }

        return length;
    }
};
```



**上面写的什么大便**

​	非常简单的反向遍历，维护一个length，当length==0时，遍历到字母+1，遍历到空格无反应；当length != 0时，遍历到字母+1，遍历到空格停止遍历

```c++
class Solution {
public:
    int lengthOfLastWord(string s) {
        int n = s.size();

        int length = 0;

        for(int i = n - 1; i >= 0; i--){
            if(length == 0){
                if(s[i] != ' '){
                    length++;
                }
            }
            else{
                if(s[i] != ' '){
                    length++;
                }
                else{
                    break;
                }
            }
        }

        return length;
    }
};
```
